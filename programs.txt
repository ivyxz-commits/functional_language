//5. Дано трёхзначное число. Выяснить, является ли оно палиндромом («перевёртышем»), т.е. таким числом
// десятичная запись которого читается одинаково слева направо и справа налево.

fn isPalindrome(n: int64) -> bool = 
    let a = n/100 in //env a = 1, n = 121
    let b = (n / 10) % 10 in 
    let c = n % 10 in 
    a == c

    let _ = print(if isPalindrome(122) then "palindrome\n" else "not palindrome\n") in
    0



//14. Найти наименьшее общее кратное двух заданных натуральных чисел.

fn gcd(a: int64, b: int64) -> int64 = 
    if b == 0 then a else gcd(b, a % b) //a - делитель

fn lcm(a: int64, b: int64) -> int64 = 
    a / gcd(a, b) * b

fn main() -> int64 =
    let _ = print(lcm(15, 60)) in
    0


//58. Найдите наибольшее значение отношения трёхзначного числа к сумме его цифр.

//сумма цифр числа
fn digitSum(n: int64) -> int64 = 
    let a = n/100 in
    let b = (n / 10) % 10 in
    let c = n % 10 in
    a + b + c

fn solve(n: int64, bestN: int64, bestNum: int64) -> int64 = 
    if n > 999 then bestN
    else 
        let s = digitSum(n) in
        let num = n / s in
        if num > bestNum
        then solve(n + 1, n, bestNum)
        else solve(n + 1, bestN, bestNum)


fn main() -> int64 =
    let best = solve(100, 100, digitSum(100)) in 
    let _ = print(best) in
    0
