data Tree = Leaf | Node(int64, Tree, Tree) //дерево рекурсивное по своей природе

fn insert(x: int64, t: Tree) -> Tree = 
    //смотрим что за дерево пришло
    match t {
        Leaf -> Node(x, Leaf, Leaf),
        Node(value, l, r) -> 
            if x < value then Node(value, insert(x, l), r)
            else if x > value then Node(value, l, insert(x, r))
            else t //x = value
    }

fn search(x: int64, t: Tree) -> string = //yep or nope
    match t {
        Leaf -> "nope\n",
        Node(value, l, r) ->
        if x == value then "yep\n" 
        else if x < value then search(x, l)
        else search(x, r)
    }

fn inorder(t: Tree) -> unit = //обход дерева в формате: лево -> корень -> право
    match t{
        Leaf -> unit,
        Node(value, l, r) -> 
            let _ = inorder(l) in
            let _ = print(value) in inorder(r) 
    }


fn main() -> int64 = let t = Leaf in let t = insert(5, t) in
    let t = insert(3, t) in let t = insert(1, t) in let t = insert(9, t) in
    let t = insert(15, t) in let _ = inorder(t) in print(search(9, t))
