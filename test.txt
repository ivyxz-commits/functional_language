fn findLocal(xs: [int64], idx: int64) -> unit =
    match xs {
        [] -> unit,
        x : [] -> print("nope"),
        x : y : [] -> print("nope"),
        x : y : rest ->
            let z = match rest {
                z : _ -> z,
            } in if y > x and y > z then
                let _ = print(idx) in
                let _ = print(idx + 1) in
                let _ = print(idx + 2) in unit
            else findLocal(y : rest, idx + 1)
    }
