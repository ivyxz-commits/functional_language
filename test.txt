//16. даны натуральные число n и вещественные числа, определить количество отрицательных 
//и количество положительных вещественных чисел

fn solve(xs: [float64], neg: int64, pos: int64) -> int64 =
    match xs {
        [] -> let _ = print(neg) in 
              let _ = print(pos) in
              0,
        x : rest ->
            if x < 0.0 
                then solve(rest, neg + 1, pos)
                else
                    if x > 0.0 
                        then solve(rest, neg, pos + 1)
                        else solve(rest, neg, pos) //x = 0.0
    }
